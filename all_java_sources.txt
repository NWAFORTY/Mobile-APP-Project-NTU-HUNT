package com.example.treasurehuntapplication;

import android.content.Context;
import android.content.SharedPreferences;

/**
 * Tracks which treasure the user is currently hunting (after accepting a clue on the treasure screen).
 * Stored per signed-in user so accounts do not share hunt state on the same device.
 */
public final class ActiveTreasureManager {

    // --- Sentinel & preference namespace ---
    private static final int NONE = -1;
    private static final String PREF = "active_treasure";
    private static final String KEY_PREFIX = "active_id_";

    private ActiveTreasureManager() {
    }

    // --- Key per user (email-based) ---

    private static SharedPreferences prefs(Context context) {
        return context.getApplicationContext().getSharedPreferences(PREF, Context.MODE_PRIVATE);
    }

    private static String safeKey(String raw) {
        return raw.trim().toLowerCase().replaceAll("[^a-z0-9]+", "_");
    }

    private static String keyForUser(Context context) {
        String email = UserSession.getEmail(context.getApplicationContext());
        if (email == null || email.isEmpty()) {
            return KEY_PREFIX + "guest";
        }
        return KEY_PREFIX + safeKey(email);
    }

    // --- Active hunt id ---

    public static void setActiveTreasure(Context context, int id) {
        prefs(context).edit().putInt(keyForUser(context), id).apply();
    }

    public static void clearActiveTreasure(Context context) {
        prefs(context).edit().remove(keyForUser(context)).apply();
    }

    public static int getActiveTreasureId(Context context) {
        String k = keyForUser(context);
        if (!prefs(context).contains(k)) {
            return NONE;
        }
        return prefs(context).getInt(k, NONE);
    }

    public static boolean hasActiveTreasure(Context context) {
        return getActiveTreasureId(context) != NONE;
    }

    public static Treasure getActiveTreasure(Context context) {
        int id = getActiveTreasureId(context);
        if (id == NONE) {
            return null;
        }
        return TreasureManager.getById(id);
    }

    public static boolean isActive(Context context, int id) {
        return getActiveTreasureId(context) == id;
    }

    // --- Reset (all users on device) ---

    /** Clears active-hunt state for every account on this device. */
    public static void clearAllActiveTreasureState(Context context) {
        prefs(context).edit().clear().apply();
    }
}

package com.example.treasurehuntapplication;

import android.util.Patterns;

/**
 * Normalizes and validates login and signup fields.
 * <p>
 * Firebase email flows require Nottingham Trent University student addresses: the substring after
 * the final {@code @} must equal {@link #REQUIRED_EMAIL_DOMAIN} (personal domains such as
 * {@code gmail.com} or {@code outlook.com}, staff domains like {@code ntu.ac.uk} without {@code my.},
 * and subdomains such as {@code mail.my.ntu.ac.uk} are rejected).
 */
public final class AuthInputValidator {

    public static final int MIN_PASSWORD_LENGTH = 6;

    /**
     * Host part (domain) that {@link #isAllowedNtuLoginEmail} requires after the last {@code @}.
     * Compares against {@link #normalizeEmail(String)} output, so matching is case-insensitive.
     */
    public static final String REQUIRED_EMAIL_DOMAIN = "my.ntu.ac.uk";

    private AuthInputValidator() {
    }

    public static String normalizeEmail(String raw) {
        if (raw == null) {
            return "";
        }
        return raw.trim().toLowerCase();
    }

    public static boolean isValidEmail(String email) {
        return email != null && !email.isEmpty() && Patterns.EMAIL_ADDRESS.matcher(email).matches();
    }

    /**
     * Whether {@code email} may be used for Firebase sign-in, password reset, or registration.
     * <ul>
     *   <li>Must pass {@link #isValidEmail(String)} (Android {@link android.util.Patterns#EMAIL_ADDRESS}).</li>
     *   <li>After {@link #normalizeEmail(String)}, the substring after the <em>last</em> {@code @} must equal
     *       {@link #REQUIRED_EMAIL_DOMAIN} exactly (case-insensitive because of normalization).</li>
     * </ul>
     * Examples: {@code alice@my.ntu.ac.uk} and {@code Bob@MY.NTU.AC.UK} allowed;
     * {@code x@gmail.com}, {@code x@outlook.com}, {@code x@ntu.ac.uk} (no {@code my.}), and
     * {@code x@mail.my.ntu.ac.uk} (subdomain, not exact host) are not allowed.
     * {@code lastIndexOf('@')} handles local parts that themselves contain {@code @}.
     */
    public static boolean isAllowedNtuLoginEmail(String email) {
        if (!isValidEmail(email)) {
            return false;
        }
        String e = normalizeEmail(email);
        int at = e.lastIndexOf('@');
        if (at < 1) {
            return false;
        }
        return REQUIRED_EMAIL_DOMAIN.equals(e.substring(at + 1));
    }

    public static boolean isValidPasswordLength(String password) {
        return password != null && password.length() >= MIN_PASSWORD_LENGTH;
    }
}

package com.example.treasurehuntapplication;

import android.location.Location;

import com.google.android.gms.maps.model.LatLng;

/** Default map camera for NTU Clifton area. */
public final class CampusMapDefaults {

    public static final float ZOOM_OVERVIEW = 16f;

    private static final LatLng CLIFTON_CAMPUS = new LatLng(52.9115, -1.1865);
    /** Rough radius (m) within which we prefer the device location over the campus default. */
    private static final float NEAR_CAMPUS_METERS = 25_000f;

    private CampusMapDefaults() {
    }

    public static LatLng cliftonCampus() {
        return CLIFTON_CAMPUS;
    }

    public static boolean shouldUseDeviceLocationForCamera(Location location) {
        if (location == null) {
            return false;
        }
        float[] out = new float[1];
        Location.distanceBetween(
                location.getLatitude(),
                location.getLongitude(),
                CLIFTON_CAMPUS.latitude,
                CLIFTON_CAMPUS.longitude,
                out);
        return out[0] <= NEAR_CAMPUS_METERS;
    }
}

package com.example.treasurehuntapplication;

import android.content.Context;

import androidx.annotation.Nullable;

import com.google.firebase.auth.FirebaseAuth;
import com.google.firebase.auth.FirebaseUser;

/**
 * Single place for {@code users/{documentId}} ids so profile, leaderboard sync, and game state
 * always target the same Firestore document (normalized email).
 */
public final class FirestoreUserDoc {

    private FirestoreUserDoc() {
    }

    /** Document id from a Firebase user, or null if missing. */
    @Nullable
    public static String idForFirebaseUser(@Nullable FirebaseUser u) {
        if (u == null || u.getEmail() == null || u.getEmail().isEmpty()) {
            return null;
        }
        return AuthInputValidator.normalizeEmail(u.getEmail());
    }

    /** Document id from {@link UserSession}, or null if not signed in. */
    @Nullable
    public static String idForSession(Context context) {
        String email = UserSession.getEmail(context.getApplicationContext());
        if (email == null || email.isEmpty()) {
            return null;
        }
        return AuthInputValidator.normalizeEmail(email);
    }

    /**
     * Prefer the signed-in Firebase account’s email, else session email (e.g. offline flows).
     */
    @Nullable
    public static String idForCurrentUser(Context context) {
        String fromAuth = idForFirebaseUser(FirebaseAuth.getInstance().getCurrentUser());
        if (fromAuth != null) {
            return fromAuth;
        }
        return idForSession(context);
    }
}

package com.example.treasurehuntapplication;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.graphics.Path;
import android.os.Build;
import android.util.AttributeSet;
import android.view.View;

import androidx.appcompat.widget.AppCompatImageView;

/**
 * {@link AppCompatImageView} clipped to a regular hexagon, with an optional stroke border.
 */
public class HexagonImageView extends AppCompatImageView {

    private final Path hexPath = new Path();
    private float strokeWidthPx;
    private int strokeColor = Color.GRAY;
    private final Paint strokePaint = new Paint(Paint.ANTI_ALIAS_FLAG);

    public HexagonImageView(Context context) {
        super(context);
        init();
    }

    public HexagonImageView(Context context, AttributeSet attrs) {
        super(context, attrs);
        init();
    }

    public HexagonImageView(Context context, AttributeSet attrs, int defStyleAttr) {
        super(context, attrs, defStyleAttr);
        init();
    }

    private void init() {
        setScaleType(ScaleType.CENTER_CROP);
        strokePaint.setStyle(Paint.Style.STROKE);
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) {
            setLayerType(View.LAYER_TYPE_SOFTWARE, null);
        }
    }

    @Override
    protected void onSizeChanged(int w, int h, int oldw, int oldh) {
        super.onSizeChanged(w, h, oldw, oldh);
        buildHexPath(w, h);
    }

    private void buildHexPath(int w, int h) {
        hexPath.reset();
        float cx = w / 2f;
        float cy = h / 2f;
        float inset = Math.max(0f, strokeWidthPx / 2f);
        float r = Math.min(w, h) / 2f - inset;
        if (r <= 0f) {
            return;
        }
        for (int i = 0; i < 6; i++) {
            double angle = Math.PI / 3 * i - Math.PI / 6;
            float x = (float) (cx + r * Math.cos(angle));
            float y = (float) (cy + r * Math.sin(angle));
            if (i == 0) {
                hexPath.moveTo(x, y);
            } else {
                hexPath.lineTo(x, y);
            }
        }
        hexPath.close();
    }

    public void setStrokeWidthDp(float dp) {
        strokeWidthPx = dp * getResources().getDisplayMetrics().density;
        strokePaint.setStrokeWidth(strokeWidthPx);
        buildHexPath(getWidth(), getHeight());
        invalidate();
    }

    public void setStrokeColor(int color) {
        strokeColor = color;
        strokePaint.setColor(color);
        invalidate();
    }

    @Override
    protected void onDraw(Canvas canvas) {
        if (hexPath.isEmpty()) {
            buildHexPath(getWidth(), getHeight());
        }
        canvas.save();
        canvas.clipPath(hexPath);
        super.onDraw(canvas);
        canvas.restore();
        strokePaint.setColor(strokeColor);
        strokePaint.setStrokeWidth(strokeWidthPx);
        canvas.drawPath(hexPath, strokePaint);
    }
}

                onPlayerClickListener.onPlayerClick(user);

package com.example.treasurehuntapplication;

import android.os.Bundle;
import android.util.Log;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.TextView;
import android.widget.Toast;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;
import androidx.recyclerview.widget.LinearLayoutManager;
import androidx.recyclerview.widget.RecyclerView;

import com.google.firebase.firestore.DocumentSnapshot;
import com.google.firebase.firestore.FirebaseFirestore;
import com.google.firebase.firestore.ListenerRegistration;
import com.google.firebase.firestore.QuerySnapshot;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

/**
 * Live Firestore leaderboard: listens to {@code users} collection, sorts by score, shows top N,
 * empty state, and opens {@link PublicPlayerStatsDialog} on row tap.
 */
public class LeaderboardFragment extends Fragment {

    private static final String TAG = "LeaderboardFragment";
    private static final int MAX_FETCH = 200;
    private static final int TOP_N = 50;

    private RecyclerView recyclerView;
    private TextView emptyView;
    private LeaderboardAdapter adapter;
    private List<LeaderboardModel> userList;
    private FirebaseFirestore db;
    @Nullable
    private ListenerRegistration leaderboardRegistration;

    public LeaderboardFragment() {
    }

    @Nullable
    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, Bundle savedInstanceState) {
        View view = inflater.inflate(R.layout.fragment_leaderboard, container, false);

        // --- RecyclerView + adapter ---
        userList = new ArrayList<>();
        adapter = new LeaderboardAdapter(userList);

        recyclerView = view.findViewById(R.id.leaderboardRecycler);
        emptyView = view.findViewById(R.id.leaderboardEmpty);
        recyclerView.setLayoutManager(new LinearLayoutManager(requireContext()));
        recyclerView.setAdapter(adapter);
        adapter.setOnPlayerClickListener(user -> PublicPlayerStatsDialog.show(LeaderboardFragment.this, user));

        db = FirebaseFirestore.getInstance();

        return view;
    }

    @Override
    public void onStart() {
        super.onStart();
        // --- Subscribe to Firestore (active while fragment started) ---
        attachLeaderboardListener();
    }

    @Override
    public void onStop() {
        detachLeaderboardListener();
        super.onStop();
    }

    /** Removes snapshot listener to avoid leaks and wasted reads. */
    private void detachLeaderboardListener() {
        if (leaderboardRegistration != null) {
            leaderboardRegistration.remove();
            leaderboardRegistration = null;
        }
    }

    /** Attaches listener to {@code users} with limit; handles errors and drives {@link #applyLeaderboardDocuments}. */
    private void attachLeaderboardListener() {
        detachLeaderboardListener();

        leaderboardRegistration = db.collection("users")
                .limit(MAX_FETCH)
                .addSnapshotListener((QuerySnapshot snapshot, com.google.firebase.firestore.FirebaseFirestoreException error) -> {
                    if (!isAdded()) {
                        return;
                    }
                    if (error != null) {
                        Log.e(TAG, "Leaderboard listener failed", error);
                        if (getContext() != null) {
                            String msg = error.getMessage() != null ? error.getMessage() : "Unknown error";
                            Toast.makeText(getContext(), getString(R.string.leaderboard_error, msg), Toast.LENGTH_LONG).show();
                        }
                        updateEmptyState();
                        return;
                    }
                    if (snapshot == null) {
                        return;
                    }
                    applyLeaderboardDocuments(snapshot);
                });
    }

    /** Parses documents, sorts descending by score, trims to TOP_N. */
    private void applyLeaderboardDocuments(QuerySnapshot snapshot) {
        userList.clear();
        for (DocumentSnapshot document : snapshot.getDocuments()) {
            userList.add(LeaderboardModel.fromDocument(document));
        }

        Collections.sort(userList, (a, b) -> Long.compare(b.getScore(), a.getScore()));
        if (userList.size() > TOP_N) {
            List<LeaderboardModel> top = new ArrayList<>(userList.subList(0, TOP_N));
            userList.clear();
            userList.addAll(top);
        }

        adapter.notifyDataSetChanged();
        updateEmptyState();
    }

    /** Toggles placeholder vs list when there are zero rows. */
    private void updateEmptyState() {
        if (emptyView == null || recyclerView == null) {
            return;
        }
        boolean empty = userList.isEmpty();
        emptyView.setVisibility(empty ? View.VISIBLE : View.GONE);
        recyclerView.setVisibility(empty ? View.GONE : View.VISIBLE);
    }
}

package com.example.treasurehuntapplication;

import androidx.annotation.NonNull;

import com.google.firebase.firestore.DocumentSnapshot;

/**
 * Firestore user fields used on the leaderboard.
 * Documents use {@code username} (display) and {@code score}; {@code email} is a fallback for older docs.
 */
public class LeaderboardModel {

    // --- Row fields (from Firestore or local demo) ---
    private String username;
    private String email;
    private long score;
    /** Optional subtitle under the display name (e.g. campus line from profile settings). */
    private String profileLocation;
    /** Firestore document id (used to load full profile); may differ from email in edge cases. */
    private String firestoreDocumentId;
    /** Firebase Auth uid when present — matches the signed-in user even if email fields differ. */
    private String firebaseUid;

    public LeaderboardModel() {
    }

    public LeaderboardModel(String username, long score) {
        this.username = username;
        this.score = score;
    }

    // --- Firestore mapping ---

    /**
     * Reads only leaderboard fields. Avoids {@link DocumentSnapshot#toObject(Class)} so extra
     * document keys (e.g. legacy {@code password}, nested profile maps) do not spam the mapper.
     */
    @NonNull
    public static LeaderboardModel fromDocument(@NonNull DocumentSnapshot document) {
        LeaderboardModel user = new LeaderboardModel();
        if (document.contains("score")) {
            Long s = document.getLong("score");
            user.setScore(s != null ? s : 0L);
        }
        String docEmail = document.getString("email");
        if (docEmail == null || docEmail.isEmpty()) {
            docEmail = document.getId();
        }
        user.setEmail(docEmail);
        String u = document.getString("username");
        user.setUsername(u != null && !u.isEmpty() ? u : docEmail);
        String loc = document.getString("profileLocation");
        user.setProfileLocation(loc != null ? loc : "");
        String uid = document.getString("uid");
        user.setFirebaseUid(uid != null ? uid : "");
        user.setFirestoreDocumentId(document.getId());
        return user;
    }

    // --- Accessors ---

    public String getFirestoreDocumentId() {
        if (firestoreDocumentId != null && !firestoreDocumentId.isEmpty()) {
            return firestoreDocumentId;
        }
        if (email != null && !email.isEmpty()) {
            return email.trim();
        }
        return "";
    }

    public void setFirestoreDocumentId(String firestoreDocumentId) {
        this.firestoreDocumentId = firestoreDocumentId;
    }

    public String getUsername() {
        return username;
    }

    public String getEmail() {
        return email;
    }

    public long getScore() {
        return score;
    }

    public void setScore(long score) {
        this.score = score;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public void setUsername(String username) {
        this.username = username;
    }

    public String getProfileLocation() {
        return profileLocation != null ? profileLocation : "";
    }

    public void setProfileLocation(String profileLocation) {
        this.profileLocation = profileLocation != null ? profileLocation : "";
    }

    public String getFirebaseUid() {
        return firebaseUid != null ? firebaseUid : "";
    }

    public void setFirebaseUid(String firebaseUid) {
        this.firebaseUid = firebaseUid != null ? firebaseUid : "";
    }

    /** Prefer username; fall back to email for legacy documents that only stored email. */
    public String getDisplayName() {
        if (username != null && !username.trim().isEmpty()) {
            return username;
        }
        if (email != null && !email.trim().isEmpty()) {
            return email;
        }
        return "Player";
    }
}

package com.example.treasurehuntapplication;

import android.content.Context;

import com.google.firebase.auth.FirebaseAuth;
import com.google.firebase.auth.FirebaseUser;
import com.google.firebase.firestore.FirebaseFirestore;
import com.google.firebase.firestore.SetOptions;

import java.util.HashMap;
import java.util.Map;

/** Ensures a {@code users/{email}} row exists for Firebase-signed-in accounts. */
public final class LeaderboardUserRegistration {

    private LeaderboardUserRegistration() {
    }

    public static void ensureRegistered(Context context) {
        FirebaseUser u = FirebaseAuth.getInstance().getCurrentUser();
        if (u == null || u.getEmail() == null || u.getEmail().isEmpty()) {
            return;
        }
        String docId = FirestoreUserDoc.idForFirebaseUser(u);
        if (docId == null) {
            return;
        }
        FirebaseFirestore.getInstance()
                .collection("users")
                .document(docId)
                .get()
                .addOnSuccessListener(doc -> {
                    if (!doc.exists()) {
                        Map<String, Object> initial = new HashMap<>();
                        initial.put("uid", u.getUid());
                        initial.put("email", docId);
                        initial.put("username", docId);
                        initial.put("score", ScoreRepository.getLocalScore(context.getApplicationContext()));
                        FirebaseFirestore.getInstance()
                                .collection("users")
                                .document(docId)
                                .set(initial, SetOptions.merge());
                        return;
                    }
                    String existingUid = doc.getString("uid");
                    if (existingUid == null || existingUid.isEmpty()) {
                        Map<String, Object> patch = new HashMap<>();
                        patch.put("uid", u.getUid());
                        FirebaseFirestore.getInstance()
                                .collection("users")
                                .document(docId)
                                .set(patch, SetOptions.merge());
                    }
                });
    }

                    String signedEmail = result.getUser() != null && result.getUser().getEmail() != null

                        .addOnSuccessListener(result -> {
                            // Migration path: new Auth user must still be @my.ntu.ac.uk; else remove stray account.
                            String signedEmail = result.getUser() != null && result.getUser().getEmail() != null
                                    ? result.getUser().getEmail()
                                    : email;
                            if (!AuthInputValidator.isAllowedNtuLoginEmail(signedEmail)) {
                                if (result.getUser() != null) {
                                    result.getUser().delete();
                                }
                                auth.signOut();
                                UserSession.signOut(Login.this);
                                Toast.makeText(Login.this, R.string.auth_email_ntu_only, Toast.LENGTH_LONG).show();
                                return;
                            }

                                UserSession.signIn(Login.this, result.getUser().getEmail());

                                goToMain();
                            }
                        }));
    }

package com.example.treasurehuntapplication;

import android.app.Activity;
import android.content.Context;
import android.content.SharedPreferences;
import android.widget.CheckBox;
import android.widget.EditText;

/** Persists “remember me” (username only) for the login form. */
public final class LoginRememberPreferences {

    private static final String PREF = "login_remember";
    private static final String KEY_REMEMBER = "remember";
    private static final String KEY_USERNAME = "username";

    private LoginRememberPreferences() {
    }

    private static SharedPreferences prefs(Context context) {
        return context.getApplicationContext().getSharedPreferences(PREF, Context.MODE_PRIVATE);
    }

    public static void loadIntoForm(Activity activity, EditText usernameInput, CheckBox rememberCheck) {
        SharedPreferences p = prefs(activity);
        rememberCheck.setChecked(p.getBoolean(KEY_REMEMBER, false));
        String u = p.getString(KEY_USERNAME, "");
        if (u != null && !u.isEmpty()) {
            usernameInput.setText(u);
        }
    }

    public static void save(Context context, boolean remember, String username) {
        SharedPreferences.Editor e = prefs(context).edit();
        if (remember && username != null && !username.trim().isEmpty()) {
            e.putBoolean(KEY_REMEMBER, true);
            e.putString(KEY_USERNAME, username.trim());
        } else {
            e.putBoolean(KEY_REMEMBER, false);
            e.remove(KEY_USERNAME);
        }
        e.apply();
    }

    public static void clear(Context context) {
        prefs(context).edit().clear().apply();
    }
}

            return true;
        }
        if (!ActiveTreasureManager.isActive(requireContext(), treasureId)) {
            return true;
        }
        if (lastUserLocation == null) {
            Toast.makeText(requireContext(), R.string.map_wait_for_location, Toast.LENGTH_SHORT).show();
            return true;
        }
        float[] d = new float[1];
        Location.distanceBetween(
                lastUserLocation.getLatitude(),
                lastUserLocation.getLongitude(),
                treasure.getLatitude(),
                treasure.getLongitude(),
                d);
        if (d[0] > COLLECT_TAP_METERS) {
            Toast.makeText(requireContext(), R.string.map_too_far_to_collect, Toast.LENGTH_SHORT).show();
            return true;
        }

        NtuMaterialDialogs.builder(requireContext())
                .setTitle(R.string.map_collect_title)
                .setMessage(getString(R.string.map_collect_message, treasure.getTitle(), treasure.getPoints()))
                .setPositiveButton(R.string.map_collect_confirm, (dialog, which) -> onTreasureCollected(treasure))
                .setNegativeButton(android.R.string.cancel, null)
                .show();
        return true;
    }

    /** Periodic GPS updates while map is visible (walk distance + marker refresh). */
    private void maybeStartProximityUpdates() {
        if (!hasLocationPermission() || mMap == null) {
            return;
        }
        Treasure active = ActiveTreasureManager.getActiveTreasure(requireContext());
        if (active == null || TreasureManager.isFound(requireContext(), active.getId())) {
            stopProximityUpdates();
            return;
        }
        if (locationUpdatesActive) {
            return;
        }
        LocationRequest request = new LocationRequest.Builder(Priority.PRIORITY_HIGH_ACCURACY, 3_000L)
                .setMinUpdateIntervalMillis(2_000L)
                .setMaxUpdateDelayMillis(8_000L)
                .build();
        try {
            fusedLocationClient.requestLocationUpdates(request, proximityCallback, Looper.getMainLooper());
            locationUpdatesActive = true;
        } catch (SecurityException e) {
            locationUpdatesActive = false;
        }
    }

    /** Stops GPS updates when fragment pauses to save battery. */
    private void stopProximityUpdates() {
        if (!locationUpdatesActive) {
            return;
        }
        try {
            fusedLocationClient.removeLocationUpdates(proximityCallback);
        } catch (SecurityException ignored) {
        }
        locationUpdatesActive = false;
    }

    /** Persists found state, points, clears active hunt, syncs Firestore, refreshes map. */
    private void onTreasureCollected(@NonNull Treasure treasure) {
        TreasureManager.markFound(requireContext(), treasure.getId());
        ActiveTreasureManager.clearActiveTreasure(requireContext());
        ScoreRepository.addPoints(requireContext(), treasure.getPoints());
        UserGameStateFirestoreSync.pushGameStateToFirestore(requireContext());
        Toast.makeText(requireContext(),
                getString(R.string.toast_treasure_found, treasure.getPoints()),
                Toast.LENGTH_LONG).show();
        stopProximityUpdates();
        refreshTreasureMarkers();
    }
}

    private void setBadgeImage(@NonNull ImageView target, int drawableRes) {
        Bitmap cached = BADGE_BITMAP_CACHE.get(drawableRes);
        if (cached != null && !cached.isRecycled()) {
            target.setImageBitmap(cached);
            return;
        }

        BitmapFactory.Options bounds = new BitmapFactory.Options();
        bounds.inJustDecodeBounds = true;
        BitmapFactory.decodeResource(getResources(), drawableRes, bounds);

        BitmapFactory.Options opts = new BitmapFactory.Options();
        opts.inPreferredConfig = Bitmap.Config.RGB_565;
        opts.inDither = true;
        opts.inSampleSize = computeInSampleSize(bounds, BADGE_TARGET_PX, BADGE_TARGET_PX);

        Bitmap bmp = BitmapFactory.decodeResource(getResources(), drawableRes, opts);
        if (bmp != null) {
            BADGE_BITMAP_CACHE.put(drawableRes, bmp);
            target.setImageBitmap(bmp);
        } else {
            target.setImageResource(drawableRes);
        }
    }

    private static int computeInSampleSize(@NonNull BitmapFactory.Options bounds, int reqWidth, int reqHeight) {
        int height = Math.max(1, bounds.outHeight);
        int width = Math.max(1, bounds.outWidth);
        int inSampleSize = 1;

        if (height > reqHeight || width > reqWidth) {
            int halfHeight = height / 2;
            int halfWidth = width / 2;

            while ((halfHeight / inSampleSize) >= reqHeight && (halfWidth / inSampleSize) >= reqWidth) {
                inSampleSize *= 2;
            }
        }
        return Math.max(1, inSampleSize);
    }

            return;
        }
        String docId = user.getFirestoreDocumentId();
        if (docId == null || docId.isEmpty()) {

                    String body = String.format(Locale.getDefault(),

/** Per-user score stored locally; Firestore sync is handled by {@link UserGameStateFirestoreSync}. */
public final class ScoreRepository {

    private static final String PREF = "score_repository";
    private static final String KEY_SCORE = "score_";

    private ScoreRepository() {
    }

    private static SharedPreferences prefs(Context context) {
        return context.getApplicationContext().getSharedPreferences(PREF, Context.MODE_PRIVATE);
    }

    private static String safeKey(String raw) {
        return raw.trim().toLowerCase().replaceAll("[^a-z0-9]+", "_");
    }

    private static String keyForUser(Context context) {
        String email = UserSession.getEmail(context.getApplicationContext());
        if (email == null || email.isEmpty()) {
            return KEY_SCORE + "guest";
        }
        return KEY_SCORE + safeKey(email);
    }

    public static long getLocalScore(Context context) {
        return prefs(context).getLong(keyForUser(context), 0L);
    }

    public static void addPoints(Context context, int delta) {
        if (delta <= 0) {
            return;
        }
        String k = keyForUser(context);
        long next = prefs(context).getLong(k, 0L) + delta;
        prefs(context).edit().putLong(k, next).apply();
    }

    public static void getDisplayScore(Context context, Consumer<Long> callback) {
        long score = getLocalScore(context);
        new Handler(Looper.getMainLooper()).post(() -> callback.accept(score));
    }

    public static void clearAll(Context context) {
        prefs(context).edit().clear().apply();
    }
}

/**
 * New account registration: Firebase Auth user creation, initial {@code users/{email}} document
 * in Firestore (email, username placeholder, score 0), then return to {@link Login}.
 * <p>
 * Registration is refused unless the email is a valid {@code @my.ntu.ac.uk} student address
 * ({@link AuthInputValidator#isAllowedNtuLoginEmail}), same rule as {@link Login}.
 */
public class Signup extends AppCompatActivity {

    // --- Firebase ---
    private FirebaseFirestore db;
    private FirebaseAuth auth;
    private EditText usernameInput;
    private EditText passwordInput;
    private EditText confirmPasswordInput;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_signup);

        // --- Edge-to-edge insets ---
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, windowInsets) -> {
            Insets bars = windowInsets.getInsets(WindowInsetsCompat.Type.systemBars());
            Insets cutout = windowInsets.getInsets(WindowInsetsCompat.Type.displayCutout());
            v.setPadding(
                    Math.max(bars.left, cutout.left),
                    Math.max(bars.top, cutout.top),
                    Math.max(bars.right, cutout.right),
                    Math.max(bars.bottom, cutout.bottom));
            return windowInsets;
        });

        // --- Services & fields ---
        db = FirebaseFirestore.getInstance();
        auth = FirebaseAuth.getInstance();

        usernameInput = findViewById(R.id.signup_username);
        passwordInput = findViewById(R.id.signup_password);
        confirmPasswordInput = findViewById(R.id.signup_confirm_password);
        Button signupButton = findViewById(R.id.signup_create);
        Button backToLoginButton = findViewById(R.id.signup_back);

        signupButton.setOnClickListener(v -> createAccount());
        backToLoginButton.setOnClickListener(v -> startActivity(new Intent(Signup.this, Login.class)));
    }

    /** Validates input, creates Auth user, writes Firestore profile; rolls back Auth if Firestore fails. */
    private void createAccount() {
        String email = AuthInputValidator.normalizeEmail(usernameInput.getText().toString());
        String password = passwordInput.getText().toString();
        String confirm = confirmPasswordInput.getText().toString();

        if (email.isEmpty() || password.isEmpty() || confirm.isEmpty()) {
            Toast.makeText(this, R.string.login_empty_fields, Toast.LENGTH_SHORT).show();
            return;
        }
        if (!AuthInputValidator.isValidEmail(email)) {
            Toast.makeText(this, R.string.auth_invalid_email, Toast.LENGTH_SHORT).show();
            return;
        }
        // Align with Login: no Firebase account for Gmail, Outlook, staff-only ntu.ac.uk, etc.
        if (!AuthInputValidator.isAllowedNtuLoginEmail(email)) {
            Toast.makeText(this, R.string.auth_email_ntu_only, Toast.LENGTH_LONG).show();
            return;
        }
        if (!AuthInputValidator.isValidPasswordLength(password)) {
            Toast.makeText(this, getString(R.string.auth_password_too_short,
                    AuthInputValidator.MIN_PASSWORD_LENGTH), Toast.LENGTH_LONG).show();
            return;
        }
        if (!password.equals(confirm)) {
            Toast.makeText(this, R.string.passwords_do_not_match, Toast.LENGTH_SHORT).show();
            return;
        }

        auth.createUserWithEmailAndPassword(email, password)
                .addOnSuccessListener(authResult -> {
                    FirebaseUser user = authResult.getUser();
                    String rawId = user != null && user.getEmail() != null ? user.getEmail() : email;
                    String docId = AuthInputValidator.normalizeEmail(rawId);
                    Map<String, Object> profile = new HashMap<>();
                    if (user != null) {
                        profile.put("uid", user.getUid());
                    }
                    profile.put("email", docId);
                    profile.put("username", docId);
                    profile.put("score", 0L);

                    db.collection("users").document(docId).set(profile)
                            .addOnSuccessListener(aVoid -> {
                                Toast.makeText(Signup.this, R.string.account_created, Toast.LENGTH_SHORT).show();
                                goToLogin();
                            })
                            .addOnFailureListener(e -> {
                                if (user != null) {
                                    user.delete();
                                }
                                Toast.makeText(Signup.this, R.string.auth_firestore_profile_failed,
                                        Toast.LENGTH_LONG).show();
                            });
                })
                .addOnFailureListener(e -> {
                    if (e instanceof FirebaseAuthUserCollisionException) {
                        Toast.makeText(Signup.this, R.string.auth_signup_failed, Toast.LENGTH_LONG).show();
                    } else {
                        Toast.makeText(Signup.this, R.string.auth_signup_failed, Toast.LENGTH_LONG).show();
                    }
                });
    }

    /** Returns to login without stacking duplicate activities. */
    private void goToLogin() {
        Intent i = new Intent(Signup.this, Login.class);
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TOP);
        startActivity(i);
        finish();
    }
}

    /** One scroll tile: coloured card, tap opens popup if not yet collected. */
    private void addTreasureTile(LinearLayout container, Treasure treasure) {
        MaterialCardView wrapper = new MaterialCardView(requireContext());
        LinearLayout.LayoutParams wrapParams =
                new LinearLayout.LayoutParams(500, 500);
        wrapParams.setMargins(0, 0, 20, 0);
        wrapper.setLayoutParams(wrapParams);
        float density = getResources().getDisplayMetrics().density;
        int cornerPx = Math.round(12 * density);
        int strokePx = Math.max(1, Math.round(1 * density));
        wrapper.setRadius(cornerPx);
        wrapper.setCardElevation(0f);
        wrapper.setStrokeWidth(strokePx);
        wrapper.setStrokeColor(ColorStateList.valueOf(
                ContextCompat.getColor(requireContext(), R.color.ntu_leaderboard_row_stroke)));
        wrapper.setCardBackgroundColor(tileColor(treasure.getRarity()));
        wrapper.setPreventCornerOverlap(false);
        wrapper.setUseCompatPadding(false);

        ImageView img = new ImageView(requireContext());
        FrameLayout.LayoutParams imgParams = new FrameLayout.LayoutParams(
                FrameLayout.LayoutParams.MATCH_PARENT,
                FrameLayout.LayoutParams.MATCH_PARENT);
        img.setLayoutParams(imgParams);
        boolean found = TreasureManager.isFound(requireContext(), treasure.getId());
        // Do not use setEnabled(false) when collected — that blocks long-press for "reset for testing".
        img.setAlpha(found ? 0.88f : 1f);

        if (!found) {
            img.setOnClickListener(v -> showTreasurePopup(treasure));
        }

        View.OnLongClickListener resetCollectedListener = v -> {
            if (!TreasureManager.isFound(requireContext(), treasure.getId())) {
                return false;
            }
            NtuMaterialDialogs.builder(requireContext())
                    .setTitle(R.string.treasure_reset_collected_title)
                    .setMessage(R.string.treasure_reset_collected_message)
                    .setPositiveButton(R.string.treasure_reset_collected_confirm, (d, w) -> {
                        TreasureManager.unmarkFound(requireContext(), treasure.getId());
                        View root = getView();
                        if (root != null) {
                            populateTreasures(root);
                        }
                    })
                    .setNegativeButton(R.string.settings_cancel, null)
                    .show();
            return true;
        };
        img.setLongClickable(true);
        img.setHapticFeedbackEnabled(true);
        img.setOnLongClickListener(resetCollectedListener);

        wrapper.addView(img);

        if (found) {
            TextView badge = new TextView(requireContext());
            badge.setText(R.string.treasure_collected_badge);
            badge.setTextColor(Color.WHITE);
            badge.setTypeface(null, Typeface.BOLD);
            badge.setTextSize(14f);
            badge.setGravity(Gravity.CENTER);
            badge.setBackgroundColor(0xCC000000);
            FrameLayout.LayoutParams badgeParams = new FrameLayout.LayoutParams(
                    FrameLayout.LayoutParams.MATCH_PARENT,
                    FrameLayout.LayoutParams.WRAP_CONTENT);
            badgeParams.gravity = Gravity.BOTTOM;
            badge.setPadding(8, 10, 8, 10);
            badge.setLongClickable(true);
            badge.setHapticFeedbackEnabled(true);
            badge.setOnLongClickListener(resetCollectedListener);
            wrapper.addView(badge, badgeParams);
        }

        container.addView(wrapper);
    }

                return Color.rgb(33, 150, 243);

                return Color.rgb(255, 193, 7);
            default:
                return Color.GRAY;
        }
    }

        NtuMaterialDialogs.applyCustomContentWindow(dialog, 0.7f);

package com.example.treasurehuntapplication;

import android.content.Context;
import android.content.SharedPreferences;
import android.location.Location;

/**
 * Accumulates approximate walk distance from GPS updates (per signed-in user on this device).
 */
public final class WalkDistanceTracker {

    private static final String PREF = "walk_distance";
    private static final String KEY_METERS = "meters_";
    private static final String KEY_LAT = "lat_";
    private static final String KEY_LON = "lon_";
    private static final float MAX_SEGMENT_METERS = 500f;
    private static final String KEY_TOAST_PREFIX = "walk_milestone_toast_";
    private static final int[] WALK_MILESTONES = {10, 100, 500};

    private WalkDistanceTracker() {
    }

    private static SharedPreferences prefs(Context context) {
        return context.getApplicationContext().getSharedPreferences(PREF, Context.MODE_PRIVATE);
    }

    private static String safeKey(String raw) {
        return raw.trim().toLowerCase().replaceAll("[^a-z0-9]+", "_");
    }

    private static String suffix(Context context) {
        String email = UserSession.getEmail(context.getApplicationContext());
        if (email == null || email.isEmpty()) {
            return "guest";
        }
        return safeKey(email);
    }

    public static void recordLocation(Context context, Location loc) {
        if (loc == null) {
            return;
        }
        SharedPreferences p = prefs(context);
        String s = suffix(context);
        float lastLat = p.getFloat(KEY_LAT + s, Float.NaN);
        float lastLon = p.getFloat(KEY_LON + s, Float.NaN);
        long meters = p.getLong(KEY_METERS + s, 0L);

        if (!Float.isNaN(lastLat) && !Float.isNaN(lastLon)) {
            float[] d = new float[1];
            Location.distanceBetween(lastLat, lastLon, loc.getLatitude(), loc.getLongitude(), d);
            if (d[0] > 0f && d[0] <= MAX_SEGMENT_METERS) {
                meters += (long) d[0];
            }
        }

        p.edit()
                .putLong(KEY_METERS + s, meters)
                .putFloat(KEY_LAT + s, (float) loc.getLatitude())
                .putFloat(KEY_LON + s, (float) loc.getLongitude())
                .apply();
    }

    public static long getTotalMeters(Context context) {
        return prefs(context).getLong(KEY_METERS + suffix(context), 0L);
    }

    /**
     * Records GPS movement, then returns the smallest walk milestone (10 / 100 / 500 m) that was
     * just crossed for the first time — used for a one-shot achievement toast on the map.
     */
    public static int onLocation(Context context, Location loc) {
        recordLocation(context, loc);
        long total = getTotalMeters(context);
        String s = suffix(context);
        SharedPreferences p = prefs(context);
        for (int m : WALK_MILESTONES) {
            String key = KEY_TOAST_PREFIX + m + "_" + s;
            if (total >= m && !p.getBoolean(key, false)) {
                p.edit().putBoolean(key, true).apply();
                return m;
            }
        }
        return 0;
    }

    public static void clearAll(Context context) {
        prefs(context).edit().clear().apply();
    }
}
